1. 万宗归一阵列碎尸法 (split)

这是极其恐怖且高频的重要大法！它能按你给出的「菜刀」将一整行连续的字符串全部剁成一小块一小块，并丢进一个标准的 Array 盒子里端出来。

console.log('a+very+nice+string'.split('+')); 
// 输出一个阵列：['a', 'very', 'nice', 'string'] （所有 + 号像骨架一样被抽离消解掉）

// ★ 巅峰套招：与高阶解构配合，抽人名秒杀建档 
const [firstName, lastName] = 'Jonas Schmedtmann'.split(' '); 
console.log(firstName, lastName); // 'Jonas', 'Schmedtmann'
// 用切刀（空格）瞬间一刀划作两人头！

2. 万法同源：亡魂拼合胶水 (join)

当咱们把人在数组里切碎后好好的整了容（改了大小写等等），用这个反向胶水函数，向每个数组骨节中央强行浇筑进去粘合剂，让它们首尾相连活成了一条新的巨长字符串链！

const newName = ['Mr.', firstName, lastName.toUpperCase()].join(' '); 
console.log(newName); // 'Mr. Jonas SCHMEDTMANN'
✨ 高阶整合洗礼实战：无定式长度人名一键大修王

来个女明星 'jessica ann smith davis'，把它全阵的所有头字抹上大写！我们把前面的切割术和这里的重组胶水来一次大荟萃狂欢。

const capitalizeName = function(name) {
  const names = name.split(' '); // 切成独立碎末阵列！
  const namesUpper = []; // 准备骨灰盒收集新重塑成果
  
  for (const n of names) { // 开始在血池迭代！
    // 方案1：切刀生拼硬缝 
    // namesUpper.push(n[0].toUpperCase() + n.slice(1));
    
    // 方案2：使用黑科技置换大法 replace() (高段位黑客推崇)
    namesUpper.push(n.replace(n[0], n[0].toUpperCase())); 
  }
  
  // 拼合亡魂大军并对外丢还完整尸体文本！
  console.log(namesUpper.join(' ')); 
};

capitalizeName('jessica ann smith davis'); // 'Jessica Ann Smith Davis'

3. 缺口强硬补齐法 (padStart / padEnd)

经常在填满一些固定格式要求位数的字符里，如果客户只输入了三个字，但系统非要逼它满 25 长度位数！那这招就是往里猛塞防震泡沫垫（你自选的填充符号）。

const message = 'Go to gate 23!';

// 往头部死命硬顶填充满，务必撑长至总体格拥有 25 个占位长
console.log(message.padStart(25, '+')); 
// 第一个参数是“达到多长的长轴”，第二个参数是用什么符号作塞子。
// 输出：+++++++++++Go to gate 23!
console.log('Jonas'.padStart(25, '+')); 
// 输出：++++++++++++++++++++Jonas

// 接着往尾巴死命猛灌，撑填到30总长（连击链）：
console.log(message.padStart(25, '+').padEnd(30, '+'));

4. 残暴克隆流水线 (repeat)

这最简单，比如机场航班晚点刷红字霸屏大告示：

const message2 = 'Bad weather... All Departures Delayed... ';
console.log(message2.repeat(5)); 
// 不讲道理地原地生生印出连体的 5 个满红字警告狂扫输出大回环！

const planesInLine = function(n) {
  console.log(`There are ${n} planes in line ${'✈️'.repeat(n)}`);
};
planesInLine(5); // 一次生出 5 只飞机符号的狂潮！
